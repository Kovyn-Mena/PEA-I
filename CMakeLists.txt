cmake_minimum_required(VERSION 3.15)
project(PEA-i VERSION 1.0 LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(SQLITE_AMALG "${CMAKE_CURRENT_SOURCE_DIR}/src/cpp/sqlite3.c")
add_library(pea_sqlite STATIC ${SQLITE_AMALG})
target_include_directories(pea_sqlite PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src/cpp")
set_target_properties(pea_sqlite PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(WIN32)
    target_compile_definitions(pea_sqlite PRIVATE -DSQLITE_THREADSAFE=0 -DSQLITE_ENABLE_MEMSYS5 -DSQLITE_ENABLE_FTS3 -DSQLITE_ENABLE_FTS4 -DSQLITE_ENABLE_FTS5 -DSQLITE_ENABLE_JSON1 -DSQLITE_ENABLE_RTREE -DSQLITE_ENABLE_GEOPOLY)
    target_compile_options(pea_sqlite PRIVATE /wd4996 /wd4267 /wd4244 /wd4018 /wd4068)
else()
    target_compile_options(pea_sqlite PRIVATE -w)
endif()
add_executable(pea_cpp src/cpp/main.cpp)
target_link_libraries(pea_cpp PRIVATE pea_sqlite)
enable_testing()
add_executable(test_tda src/cpp/test_tda.cpp)
target_link_libraries(test_tda PRIVATE pea_sqlite)
add_test(NAME test_tda COMMAND test_tda)
if(WIN32 AND MINGW)
    target_link_options(pea_cpp PRIVATE -static-libgcc -static-libstdc++)
endif()
